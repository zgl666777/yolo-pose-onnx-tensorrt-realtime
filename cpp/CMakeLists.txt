find_package(CUDAToolkit 12 REQUIRED)
find_package(Threads REQUIRED)
find_package(OpenCV 4 REQUIRED COMPONENTS core imgproc imgcodecs videoio highgui)
set(TENSORRT_ROOT "" CACHE PATH "TensorRT 10 installation prefix")
find_path(TENSORRT_INCLUDE NvInfer.h HINTS ${TENSORRT_ROOT}/include /usr/include/x86_64-linux-gnu)
find_library(TENSORRT_LIBRARY NAMES nvinfer nvinfer_10 HINTS ${TENSORRT_ROOT}/lib ${TENSORRT_ROOT}/lib64 /usr/lib/x86_64-linux-gnu)
if(NOT TENSORRT_INCLUDE OR NOT TENSORRT_LIBRARY)
  message(FATAL_ERROR "TensorRT not found. Set -DTENSORRT_ROOT=/path/to/TensorRT-10")
endif()
add_library(pose_trt SHARED src/engine.cpp src/c_api.cpp)
target_include_directories(pose_trt PUBLIC include PRIVATE ${TENSORRT_INCLUDE})
target_link_libraries(pose_trt PRIVATE CUDA::cudart ${TENSORRT_LIBRARY} Threads::Threads)
target_compile_options(pose_trt PRIVATE "$<$<CXX_COMPILER_ID:GNU,Clang>:-Wall;-Wextra;-Wpedantic>")
set_target_properties(pose_trt PROPERTIES WINDOWS_EXPORT_ALL_SYMBOLS ON)
add_executable(pose_camera src/main.cpp)
target_include_directories(pose_camera PRIVATE include ${OpenCV_INCLUDE_DIRS})
target_link_libraries(pose_camera PRIVATE pose_trt ${OpenCV_LIBS})
add_executable(pose_geometry_check src/geometry_check.cpp)
target_include_directories(pose_geometry_check PRIVATE include ${OpenCV_INCLUDE_DIRS})
target_link_libraries(pose_geometry_check PRIVATE ${OpenCV_LIBS})
